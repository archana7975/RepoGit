        protected void Page_Load(object sender, EventArgs e)
        {
            try
            {
                //log user navigation
                if(!IsPostBack)
                {
                    Common.RecordPageHit(Request.Url.AbsolutePath, Request.Url.ToString());//, Session["UserMacAddress"].ToString());
                }
                if (Session[Constants.SessionVariableNames.ErrorMessage] != null)
                {
                    CommonCode.RedirectPageTo(Constants.URLs.ErrorHandler);
                    return;
                }

                //Code created by Neha on 28 / 06 / 2018
                string notification = SystemManager.GetSystemNotification();               
                lblnotification.Text = notification.ToLower().Equals("no notifications") ? string.Empty : notification;
                divNotification.Visible = (!lblnotification.Text.Equals(String.Empty));

                var timeoutValue = CommonCode.GetTimeOutValue();
                var timeoutValueSeconds = timeoutValue * 60;

                if (TimeoutSession.Content != timeoutValueSeconds.ToString() /*+ ";URL=../Login.aspx"*/)
                    TimeoutSession.Content = timeoutValueSeconds.ToString();// + ";URL=../Login.aspx";

                //daily inventory and client dont use this masterpage
               
                if (um.IsUserInRole(STAWMSRoles.DailyInventoryManagement) || um.IsUserInRole(STAWMSRoles.WarehousingClientUser) || ((HttpContext.Current.Session == null || HttpContext.Current.Session.Count < 1)))
                {
                    CommonCode.RedirectPageTo("../LogOff.aspx");
                    return;
                }

                myUser = new User((int)HttpContext.Current.Session["currentUserPK"]);
                appSystem mySystem = new appSystem();
                userSessionInfoCookie = new HttpCookie("UserInfo");

                if (Request.Cookies["UserInfo"] == null)
                {
                    PopulateCookie();
                }
                else
                {
                    userSessionInfoCookie = Request.Cookies["UserInfo"];

                    if ( Convert.ToInt32(userSessionInfoCookie["UserPK"]) != myUser.GetValue<int>("UserPK"))
                    {
                        PopulateCookie();
                    }
                    else
                    {
                        userSessionInfoCookie.Expires = DateTime.Now.BSTDate().AddMinutes(timeoutValue);
                        Response.Cookies.Add(userSessionInfoCookie);
                    }
                }
                
                String myScriptActualLine = "$.timeoutDialog({ timeout: " + mySystem.GetValue<Int32>("TimeoutSeconds").ToString() + ", countdown: 60, logout_redirect_url: '" + mySystem.GetValue<String>("WebsiteURL") + "/LogOff.aspx\', keep_alive_url: '" + mySystem.GetValue<String>("WebsiteURL") + "/handlers/STAajax.ashx?function=KeepAlive\', restart_on_yes: true});";

                if (!IsPostBack)
                    ltrlMenu.Text = GenerateMainMenu();

                hyperLinkVersionInfo.InnerText = "Version " + GetAssemblyVersion();

                CheckForUserGuide();


                Debug.WriteLine($"HttpContext: {HttpContext.Current != null}, " +
                    $"Session: {HttpContext.Current?.Session != null}, " +
                     $"SessionID: {HttpContext.Current?.Session?.SessionID}");
              
                //keep API alive
                RefreshApiKeepAlive();
            }
            catch (Exception exception)
            {
                HandleError(exception);
            }
        }
        private void RefreshApiKeepAlive()
        {
            try
            {
                DateTime currentTime = DateTime.Now.BSTDate();
                //keep API alive
                if (Session["LastApiRefresh"] != null)
                {
                    DateTime lastRefresh = Convert.ToDateTime(Session["LastApiRefresh"]);
                    if (currentTime.AddMinutes(-10) > lastRefresh)
                    {
                        //_schedulerManager.GetBuildings();
                        _apiAliveService.RefreshTokenIfNeededAsync().ConfigureAwait(false).GetAwaiter().GetResult();
                        Session["LastApiRefresh"] = currentTime;
                    }
                }
                else
                    Session.Add("LastApiRefresh", DateTime.Now.BSTDate());

                //SetupQuickLinks(myUser.GetValue<int>("UserPK"));
            }
            catch(Exception ex)
            {
                throw ex;
            }
        }
        protected void Page_Init(object Sender, EventArgs e)
        {
            try
            {
                var pagePermission = CommonCode.IsPageAllowedForUser(Request.Url.AbsolutePath);
                if (!(pagePermission.Value > 1))
                    HttpContext.Current.Response.Redirect("/Home/Home.aspx", true);

                //set permission level so buttons can be set
                HttpContext.Current.Session["PagePermissionLevel"] = pagePermission.Value;

                string pageName = this.ContentPlaceHolder1.Page.GetType().FullName;

                /* These session variables are only required when filtering users on the users screen or creating a new user.
                 * Search for both session variables in the STAWMS project code for more information. */

                if (pageName != "ASP.security_users_aspx" && pageName != "ASP.security_manageuser_aspx" && !(Session["fltUsersFilterSearchText"] == null && Session["GridViewSortExpression"] == null))
                {
                    if (Session["fltUsersFilterSearchText"] != null) Session["fltUsersFilterSearchText"] = null;
                    if (Session["GridViewSortExpression"] != null) Session["GridViewSortExpression"] = null;
                }

                if (pageName != "ASP.invoicing_newinvoicepage_aspx" && !(Session["InvoiceNoSetDatePicker"] == null && Session["InvoiceDivisions"] == null))
                {
                    if (Session["InvoiceNoSetDatePicker"] != null) Session["InvoiceNoSetDatePicker"] = null;
                    if (Session["InvoiceDivisions"] != null) Session["InvoiceDivisions"] = null;
                }
            }
            catch (Exception exception)
            {
                if (!(exception is ThreadAbortException))//ignore thread abort thrown by response.end
                    HandleError(exception);
            }
        }

            }
            catch (Exception exception)
            {

            }
        }

        private void PopulateCookie()
        {
            try
            {
                DataSet userSessionInfo = STADatabase.SQLStoredProcedureSelect("spUserSessionInfo", new STASqlParameter("@UserPK", myUser.PK));

                userSessionInfoCookie["HelpDeskUrl"] = userSessionInfo.Tables[0].Rows[0]["HelpDeskUrl"].ToString();
                userSessionInfoCookie["ManualUrl"] = userSessionInfo.Tables[0].Rows[0]["ManualUrl"].ToString();

        private string GetAssemblyVersion()

    /// <summary>
    /// PreRender handler hides contract-related columns for unauthorized roles
    /// </summary>
    private void GvCustomers_PreRender(object sender, EventArgs e)
    {
        try
        {
            bool canSeeContracts = !(um.IsUserInRole(STAWMSRoles.WarehouseOp)
                                     || um.IsUserInRole(STAWMSRoles.StockController));

            if (gvCustomers.HeaderRow == null)
                return;

            // Detect contract columns only once
            if (_contractColumnIndexes.Count == 0)
            {
                for (int i = 0; i < gvCustomers.HeaderRow.Cells.Count; i++)
                {
                    string header = string.Empty;

                    // If the header cell has a LinkButton (sorting), get its text
                    if (gvCustomers.HeaderRow.Cells[i].Controls.Count > 0)
                    {
                        if (gvCustomers.HeaderRow.Cells[i].Controls[0] is LinkButton lb)
                            header = lb.Text.Trim();
                    }
                    else
                    {
                        header = gvCustomers.HeaderRow.Cells[i].Text.Trim();
                    }

                    if (header == "Contract Renewal Date" ||
                        header == "Renewal Reminder Interval" ||
                        header == "Renewal Alert Email")
                    {
                        _contractColumnIndexes.Add(i);
                    }
                }
            }

            // Hide/show contract columns
            foreach (int colIndex in _contractColumnIndexes)
            {
                gvCustomers.HeaderRow.Cells[colIndex].Visible = canSeeContracts;

                foreach (GridViewRow row in gvCustomers.Rows)
                {
                    if (row.Cells.Count > colIndex)
                        row.Cells[colIndex].Visible = canSeeContracts;
                }
            }
        }
        catch (Exception ex)
        {
            STAError.HandleError(ex);
        }
    }

    private void UpdateDivisionBuildingUnit()
    {
        try
        {
            //#DEV-543. Set IsArchivedField to true. Do Not Delete the Record
            Int32 myDivisionBuildingUnitPK = Int32.Parse(hdnDivisionBuildingUnitsSelectedPK.Value);
            
            DivisionBuildingUnit myDivisionBuildingUnit = new DivisionBuildingUnit(myDivisionBuildingUnitPK);
            int BuildingUnitFK = myDivisionBuildingUnit.GetValue<Int32>("BuildingUnitFK");
            int DivisionFK = myDivisionBuildingUnit.GetValue<Int32>("DivisionFK");

            Division myDivision = new Division(DivisionFK);
            string DivisionCode = myDivision.GetValue<string>("DivisionCode");

            //Update IsArchived Field. Do Not Delete Record.
            myDivisionBuildingUnit.SetValue<Boolean>("IsArchived", true);
            myDivisionBuildingUnit.Update();

            //#DEV-910 Update Audit Log
            String strDescription = "Delete Division Building Unit. Division - " + DivisionCode + ", And Associated Reserved Division Building Bay."; 
            BuildingUnitManager.AddBuildingUnitAuditRecord(BuildingUnitFK, BuildingUnitAuditType.DivisionalBuildingUnit, strDescription, -1, -1, -1, myDivisionBuildingUnitPK, -1, -1);

            // Delete the tblDivisonBuildingUnitBays
            string deleteDivisionBuildingUnitBays = "Delete From tblDivisionBuildingUnitBays Where BuildingUnitFK = " + BuildingUnitFK + " And DivisionFK = " + DivisionFK;
            STADatabase.SQLTextScalar(deleteDivisionBuildingUnitBays, false);


            // Update tblStockItemBuildingUnit Set IsArchived Field to 1 
            string updateSqlStockItemBuildingUnit = "Update StockItemBU Set StockItemBU.IsArchived = 1" +
                                                    " From tblStockItemBuildingUnits StockItemBU" +
                                                    " INNER JOIN tblStockItems On tblStockItems.StockItemPK = StockItemBU.StockItemFK " +
                                                    " Where tblStockItems.DivisionFK = " + DivisionFK +
                                                    " And StockItemBU.BuildingUnitFK = " + BuildingUnitFK ;
            STADatabase.SQLTextScalar(updateSqlStockItemBuildingUnit, false);

            //Delete tblStockItemBuildingUnitBays
            string deleteSqlStockItemBuildingUnitBays = "Delete StockItemBUB From tblStockItemBuildingUnitBays StockItemBUB" +
                                                        " INNER JOIN tblStockItems On tblStockItems.StockItemPK = StockItemBUB.StockItemFK" +
                                                        " Where tblStockItems.DivisionFK = " + DivisionFK +
                                                        " AND StockItemBUB.BuildingUnitFK = " + BuildingUnitFK;
            STADatabase.SQLTextScalar(deleteSqlStockItemBuildingUnitBays, false);
